-- Assumptions:
-- * "Engineering's three teams" = dim_org rows with is_team = TRUE whose parent department is 'Engineering'.
-- * Attrition rate = terminations dated in FY2026 (all termination types, voluntary and involuntary)
--   divided by average daily headcount over FY2026 (from fct_headcount_daily snapshots).
-- * Fiscal year comes from dim_date.fiscal_year = 2026. Only snapshot days that exist in the

  FROM `analytics.dim_org`
  WHERE is_team

  FROM `analytics.dim_date`
  WHERE fiscal_year = 2026
),

  SELECT COUNT(DISTINCT h.snapshot_date) AS n_days

  JOIN fy_days d ON d.date_day = h.snapshot_date
),
hc AS (
  SELECT h.department_id, COUNT(*) AS headcount_days
  FROM `analytics.fct_headcount_daily` h
  JOIN fy_days d ON d.date_day = h.snapshot_date

-- Interpretation: regretted attrition rate = regretted terminations in FY2026 / average headcount in FY2026, per location.
-- Fiscal year dates come from dim_date.fiscal_year = 2026 (min/max date_day).
-- Average headcount = average of daily headcount snapshots over the FY2026 days present in fct_headcount_daily, per location.
-- Terminations include all types (voluntary and involuntary) but only those flagged regretted_flag = TRUE in the numerator.
-- Rate is not annualized; it is for the fiscal year as a whole (if FY is incomplete, it covers the elapsed portion).
-- Location for terminations is taken from fct_terminations; for headcount from fct_headcount_daily.
WITH fy AS (
  SELECT date_day
  FROM analytics.dim_date
  WHERE fiscal_year = 2026
),
term AS (
  SELECT
    t.location,
    COUNTIF(t.regretted_flag) AS regretted_terms
  FROM analytics.fct_terminations t
  JOIN fy ON fy.date_day = t.termination_date
  GROUP BY t.location
),
daily AS (
  SELECT
    h.location,
    h.snapshot_date,
    COUNT(DISTINCT h.employee_id) AS hc
  FROM analytics.fct_headcount_daily h
  JOIN fy ON fy.date_day = h.snapshot_date
  GROUP BY h.location, h.snapshot_date
),
avg_hc AS (
  SELECT
    location,
    AVG(hc) AS avg_headcount
  FROM daily
  GROUP BY location
)
SELECT
  a.location,
  COALESCE(t.regretted_terms, 0) AS regretted_terms,
  a.avg_headcount,
  SAFE_DIVIDE(COALESCE(t.regretted_terms, 0), a.avg_headcount) AS regretted_attrition_rate
FROM avg_hc a
LEFT JOIN term t ON t.location = a.location
ORDER BY a.location
